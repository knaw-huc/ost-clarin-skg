PREFIX datacite: <http://purl.org/spar/datacite/>
PREFIX silvio:   <http://www.essepuntato.it/2010/06/literalreification/>
PREFIX foaf:     <http://xmlns.com/foaf/0.1/>
PREFIX skg:      <https://w3id.org/skg-if/ontology/>
PREFIX voidPlus: <https://lenticularlens.org/voidPlus/>
PREFIX resource: <https://lenticularlens.org/voidPlus/resource/>

CONSTRUCT {
    ?s a skg:Agent, foaf:Organization ;
       datacite:hasIdentifier ?id ;
       foaf:name ?n ;
       foaf:homepage ?h .

    ?id datacite:usesIdentifierScheme ?scheme ;
        silvio:hasLiteralValue ?pid .
}
WHERE {
    {
        SELECT DISTINCT ?s WHERE {
            {
                {
                    SELECT ?cluster
                           (MIN(STR(?member)) AS ?firstMember)
                    WHERE {
                        ?cluster a voidPlus:Cluster ;
                                 voidPlus:hasItem ?member .

                        ?member a foaf:Organization .

                        FILTER NOT EXISTS {
                            ?link voidPlus:hasClusterID ?cluster .

                            FILTER NOT EXISTS {
                                ?link voidPlus:hasValidation ?validation .
                                ?validation voidPlus:hasValidationStatus
                                            resource:Accepted .
                            }
                        }
                    }
                    GROUP BY ?cluster
                }

                BIND(IRI(?firstMember) AS ?s)
            }
            UNION
            {
                ?s a foaf:Organization .

                FILTER NOT EXISTS {
                    ?anyCluster a voidPlus:Cluster ;
                                voidPlus:hasItem ?s .
                }
            }

            #FILTERS#
        }
        ORDER BY ?s
        #PAGINATION#
    }

    OPTIONAL { ?s foaf:name ?n . }
    OPTIONAL { ?s foaf:homepage ?h . }
    OPTIONAL {
        ?s datacite:hasIdentifier ?id .
        OPTIONAL { ?id datacite:usesIdentifierScheme ?scheme . }
        OPTIONAL { ?id silvio:hasLiteralValue ?pid . }
    }
}
